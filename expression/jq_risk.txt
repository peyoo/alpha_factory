#20日周期（短周期）
Variance20 = ts_var(ts_returns(CLOSE, 1), 20) * 252

Skewness20 = ts_skewness(ts_returns(CLOSE, 1), 20)

Kurtosis20 = ts_kurtosis(ts_returns(CLOSE, 1), 20)

sharpe_ratio_20 = (ts_mean(ts_returns(CLOSE, 1), 20) * 252 - 0.04) / (ts_std_dev(ts_returns(CLOSE, 1), 20) * sqrt(252))
